This guide covers deploying a Better Zap application on Cloudflare Workers with D1 as the database.

## Prerequisites

- [Wrangler CLI](https://developers.cloudflare.com/workers/wrangler/) installed
- A Cloudflare account
- Your WhatsApp API credentials (see [Configuration](/docs/getting-started/configuration))

## 1. Create the Project

```bash
npm create cloudflare@latest my-whatsapp-bot -- --template hono
cd my-whatsapp-bot
npm install better-zap @better-zap/hono hono
```

## 2. Create a D1 Database

```bash
wrangler d1 create whatsapp-logs
```

Copy the database ID into your `wrangler.toml`:

```toml title="wrangler.toml"
name = "my-whatsapp-bot"
main = "src/index.ts"
compatibility_date = "2025-01-01"

[[d1_databases]]
binding = "DB"
database_name = "whatsapp-logs"
database_id = "your-database-id"
```

## 3. Create the Database Schema

```bash
wrangler d1 execute whatsapp-logs --local --command "
CREATE TABLE IF NOT EXISTS conversations (
  id TEXT PRIMARY KEY,
  phone TEXT NOT NULL UNIQUE,
  contact_name TEXT,
  unread_count INTEGER NOT NULL DEFAULT 0,
  status TEXT NOT NULL DEFAULT 'active',
  last_message_at TEXT NOT NULL,
  last_message_preview TEXT,
  last_direction TEXT NOT NULL,
  message_count INTEGER NOT NULL DEFAULT 0,
  last_incoming_message_at TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS messages (
  id TEXT PRIMARY KEY,
  wa_message_id TEXT UNIQUE,
  conversation_id TEXT NOT NULL,
  phone TEXT NOT NULL,
  user_id TEXT,
  direction TEXT NOT NULL,
  message_type TEXT NOT NULL,
  content TEXT,
  template_name TEXT,
  status TEXT NOT NULL DEFAULT 'sent',
  error_message TEXT,
  metadata TEXT,
  sent_at TEXT NOT NULL,
  delivered_at TEXT,
  read_at TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (conversation_id) REFERENCES conversations(id)
);

CREATE INDEX IF NOT EXISTS idx_messages_conversation ON messages(conversation_id, sent_at);
CREATE INDEX IF NOT EXISTS idx_messages_wa_id ON messages(wa_message_id);
CREATE INDEX IF NOT EXISTS idx_messages_recent_outgoing ON messages(phone, direction, sent_at);
CREATE INDEX IF NOT EXISTS idx_conversations_phone ON conversations(phone);
"
```

The `wa_message_id TEXT UNIQUE` constraint is required for atomic incoming
webhook deduplication.

## 4. Implement WhatsAppLogStore

Create a D1 adapter that implements the `WhatsAppLogStore` interface:

```ts title="src/database.ts"
import type { WhatsAppLogStore } from "better-zap";

type ConversationRow = {
  id: string;
  phone: string;
  contact_name: string | null;
  unread_count: number;
  status: string;
  last_message_at: string;
  last_message_preview: string | null;
  last_direction: string;
  message_count: number;
  last_incoming_message_at: string | null;
};

type MessageRow = {
  id: string;
  conversation_id: string;
  user_id: string | null;
  phone: string;
  wa_message_id: string | null;
  direction: string;
  message_type: string;
  content: string | null;
  template_name: string | null;
  status: string;
  error_message: string | null;
  metadata: string | null;
  sent_at: string;
  delivered_at: string | null;
  read_at: string | null;
};

const statusRank = {
  sent: 1,
  delivered: 2,
  read: 3,
  failed: 4,
} as const;

function toConversation(row: ConversationRow) {
  return {
    id: row.id,
    phone: row.phone,
    contactName: row.contact_name,
    unreadCount: row.unread_count,
    status: row.status,
    lastMessageAt: row.last_message_at,
    lastMessagePreview: row.last_message_preview,
    lastDirection: row.last_direction,
    messageCount: row.message_count,
    lastIncomingMessageAt: row.last_incoming_message_at,
  };
}

function toMessage(row: MessageRow) {
  return {
    id: row.id,
    conversationId: row.conversation_id,
    userId: row.user_id,
    phone: row.phone,
    waMessageId: row.wa_message_id,
    direction: row.direction as "incoming" | "outgoing",
    messageType: row.message_type as any,
    content: row.content,
    templateName: row.template_name,
    status: row.status as "sent" | "delivered" | "read" | "failed",
    errorMessage: row.error_message,
    metadata: row.metadata ? JSON.parse(row.metadata) : undefined,
    sentAt: row.sent_at,
    deliveredAt: row.delivered_at,
    readAt: row.read_at,
  };
}

export function createD1Store(db: D1Database): WhatsAppLogStore {
  return {
    async createWhatsAppLog(params) {
      const conversationId = crypto.randomUUID();
      const messageId = crypto.randomUUID();

      await db.prepare(`
        INSERT INTO conversations (
          id, phone, contact_name, last_message_at, last_message_preview,
          last_direction, message_count, last_incoming_message_at, unread_count
        )
        VALUES (?, ?, ?, ?, ?, ?, 0, NULL, 0)
        ON CONFLICT(phone) DO NOTHING
      `).bind(
        conversationId,
        params.phone,
        params.contactName ?? null,
        params.sentAt,
        params.content,
        params.direction,
      ).run();

      const conversation = await db.prepare(
        "SELECT id FROM conversations WHERE phone = ?"
      ).bind(params.phone).first<{ id: string }>();

      const insert = await db.prepare(`
        INSERT INTO messages (
          id, wa_message_id, conversation_id, phone, user_id, direction,
          message_type, content, template_name, status, error_message, metadata,
          sent_at
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT(wa_message_id) DO NOTHING
      `).bind(
        messageId,
        params.waMessageId ?? null,
        conversation?.id ?? conversationId,
        params.phone,
        params.userId ?? null,
        params.direction,
        params.messageType,
        params.content,
        params.templateName ?? null,
        params.status,
        params.errorMessage ?? null,
        params.metadata ? JSON.stringify(params.metadata) : null,
        params.sentAt,
      ).run();

      if (insert.meta.changes === 0) {
        const existing = await db.prepare(
          "SELECT * FROM messages WHERE wa_message_id = ?"
        ).bind(params.waMessageId).first<MessageRow>();
        if (!existing) throw new Error("Duplicate message row not found");
        return { record: toMessage(existing), created: false };
      }

      await db.prepare(`
        UPDATE conversations
        SET
          contact_name = COALESCE(?, contact_name),
          last_message_at = ?,
          last_message_preview = ?,
          last_direction = ?,
          message_count = message_count + 1,
          last_incoming_message_at = ?,
          unread_count = unread_count + ?,
          updated_at = datetime('now')
        WHERE id = ?
      `).bind(
        params.contactName ?? null,
        params.sentAt,
        params.content,
        params.direction,
        params.direction === "incoming" ? params.sentAt : null,
        params.direction === "incoming" ? 1 : 0,
        conversation?.id ?? conversationId,
      ).run();

      const row = await db.prepare(
        "SELECT * FROM messages WHERE id = ?"
      ).bind(messageId).first<MessageRow>();

      if (!row) throw new Error("Inserted message row not found");
      return { record: toMessage(row), created: true };
    },

    async getMessageByWaId(waMessageId) {
      const row = await db.prepare(
        "SELECT * FROM messages WHERE wa_message_id = ?"
      ).bind(waMessageId).first<MessageRow>();
      return row ? toMessage(row) : null;
    },

    async updateWhatsAppLogByWaMessageId(waMessageId, updates) {
      const columns: string[] = [];
      const values: unknown[] = [];

      const set = (column: string, value: unknown) => {
        columns.push(`${column} = ?`);
        values.push(value);
      };

      if (updates.status) set("status", updates.status);
      if (updates.errorMessage !== undefined) set("error_message", updates.errorMessage);
      if (updates.deliveredAt !== undefined) set("delivered_at", updates.deliveredAt);
      if (updates.readAt !== undefined) set("read_at", updates.readAt);
      if (updates.metadata !== undefined) {
        set("metadata", updates.metadata ? JSON.stringify(updates.metadata) : null);
      }

      if (columns.length === 0) return;

      await db.prepare(`
        UPDATE messages SET ${columns.join(", ")} WHERE wa_message_id = ?
      `).bind(...values, waMessageId).run();
    },

    updateStatusIfProgressed: async (waMessageId, newStatus, updates) => {
      const nowRank = statusRank[newStatus];
      const result = await db.prepare(`
        UPDATE messages
        SET
          status = ?,
          delivered_at = COALESCE(?, delivered_at),
          read_at = COALESCE(?, read_at),
          error_message = COALESCE(?, error_message),
          metadata = COALESCE(?, metadata)
        WHERE wa_message_id = ?
          AND ? > CASE status
            WHEN 'sent' THEN 1
            WHEN 'delivered' THEN 2
            WHEN 'read' THEN 3
            WHEN 'failed' THEN 4
            ELSE 0
          END
      `).bind(
        newStatus,
        updates.deliveredAt ?? null,
        updates.readAt ?? null,
        updates.errorMessage ?? null,
        updates.metadata ? JSON.stringify(updates.metadata) : null,
        waMessageId,
        nowRank,
      ).run();

      return result.meta.changes > 0;
    },

    async getConversationById(conversationId) {
      const row = await db.prepare(
        "SELECT * FROM conversations WHERE id = ?"
      ).bind(conversationId).first<ConversationRow>();
      return row ? toConversation(row) : null;
    },

    async getConversationByPhone(phone) {
      const row = await db.prepare(
        "SELECT * FROM conversations WHERE phone = ?"
      ).bind(phone).first<ConversationRow>();
      return row ? toConversation(row) : null;
    },

    async getConversations() {
      const { results } = await db.prepare(
        "SELECT * FROM conversations ORDER BY last_message_at DESC"
      ).all<ConversationRow>();
      return results.map(toConversation);
    },

    async getMessagesByConversationPaginated(conversationId, cursor, limit = 50) {
      let query = "SELECT * FROM messages WHERE conversation_id = ?";
      const bindings: any[] = [conversationId];

      if (cursor) {
        query += " AND sent_at < (SELECT sent_at FROM messages WHERE id = ?)";
        bindings.push(cursor);
      }

      query += " ORDER BY sent_at DESC LIMIT ?";
      bindings.push(limit);

      const { results } = await db.prepare(query).bind(...bindings).all<MessageRow>();
      return results.map((row) => ({
        id: row.id,
        phone: row.phone,
        content: row.content,
        direction: row.direction,
        status: row.status,
        sentAt: row.sent_at,
        templateName: row.template_name,
        messageType: row.message_type,
        metadata: row.metadata ? JSON.parse(row.metadata) : undefined,
      }));
    },

    async hasRecentOutgoingMessage(phone, withinHours) {
      const since = new Date(Date.now() - withinHours * 60 * 60 * 1000).toISOString();
      const row = await db.prepare(`
        SELECT 1 FROM messages
        WHERE phone = ? AND direction = 'outgoing' AND sent_at >= ?
        LIMIT 1
      `).bind(phone, since).first();

      return !!row;
    },
  };
}
```

## 5. Wire It Up

```ts title="src/index.ts"
import { betterZap } from "@better-zap/hono";
import { createD1Store } from "./database";

type Env = {
  DB: D1Database;
  WHATSAPP_TOKEN: string;
  WHATSAPP_PHONE_ID: string;
  WEBHOOK_VERIFY_TOKEN: string;
  WHATSAPP_APP_SECRET: string;
};

export default {
  fetch(request: Request, env: Env, ctx: ExecutionContext) {
    const zap = betterZap({
      database: { whatsappLog: createD1Store(env.DB) },
      config: {
        token: env.WHATSAPP_TOKEN,
        phoneId: env.WHATSAPP_PHONE_ID,
        webhookToken: env.WEBHOOK_VERIFY_TOKEN,
        appSecret: env.WHATSAPP_APP_SECRET,
      },
      webhook: {
        onMessage: async (ctx) => {
          await ctx.api.sendText(ctx.phone, `Echo: ${ctx.content}`);
        },
        onStatusUpdate: async (ctx) => {
          console.log(`${ctx.status.id}: ${ctx.status.status}`);
        },
      },
    });

    return zap.handler(request, env, ctx);
  },
};
```

## 6. Deploy

```bash
# Set secrets
wrangler secret put WHATSAPP_TOKEN
wrangler secret put WHATSAPP_PHONE_ID
wrangler secret put WEBHOOK_VERIFY_TOKEN
wrangler secret put WHATSAPP_APP_SECRET

# Apply migrations to production D1
wrangler d1 execute whatsapp-logs --remote --file schema.sql

# Deploy
wrangler deploy
```

## 7. Configure the Webhook

Set your webhook URL in [Meta's App Dashboard](https://developers.facebook.com/apps/):

```
https://my-whatsapp-bot.your-subdomain.workers.dev/api/whatsapp/webhook
```

Use the same verify token you set as `WEBHOOK_VERIFY_TOKEN`.
